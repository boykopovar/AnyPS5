function(anyps5_default_test_timeout label outVariable)
    if(label STREQUAL "unit")
        set(${outVariable} ${ANYPS5_UNIT_TEST_TIMEOUT} PARENT_SCOPE)
    else()
        set(${outVariable} "" PARENT_SCOPE)
    endif()
endfunction()

function(anyps5_set_test_properties name label timeout)
    set(properties LABELS ${label})
    if(timeout)
        list(APPEND properties TIMEOUT ${timeout})
    endif()
    set_tests_properties(${name} PROPERTIES ${properties} ${ARGN})
endfunction()

function(anyps5_require_test_label label)
    if(NOT label IN_LIST ANYPS5_TEST_LABELS)
        message(FATAL_ERROR "Unknown test label '${label}', expected one of: ${ANYPS5_TEST_LABELS}")
    endif()
endfunction()

function(anyps5_add_test target)
    cmake_parse_arguments(PARSE_ARGV 1 test "CUSTOM_MAIN" "NAME;LABEL;TIMEOUT" "SOURCES;LINK;INCLUDES;ARGS;PROPERTIES")
    if(NOT test_LABEL)
        message(FATAL_ERROR "anyps5_add_test(${target}) needs a LABEL")
    endif()
    anyps5_require_test_label(${test_LABEL})
    if(NOT test_NAME)
        string(REGEX REPLACE "_tests$" "" test_NAME "${target}")
    endif()
    if(NOT test_TIMEOUT)
        anyps5_default_test_timeout(${test_LABEL} test_TIMEOUT)
    endif()
    if(NOT test_CUSTOM_MAIN)
        list(APPEND test_SOURCES ${ANYPS5_TESTING_MAIN})
    endif()
    add_test_executable(${target} ${test_SOURCES})
    target_link_libraries(${target} PRIVATE anyps5_testing ${test_LINK})
    if(test_INCLUDES)
        target_include_directories(${target} PRIVATE ${test_INCLUDES})
    endif()
    add_test(NAME ${test_NAME} COMMAND ${target} ${test_ARGS})
    anyps5_set_test_properties(${test_NAME} ${test_LABEL} "${test_TIMEOUT}" SKIP_RETURN_CODE 77 ${test_PROPERTIES})
endfunction()

function(anyps5_add_python_test name)
    cmake_parse_arguments(PARSE_ARGV 1 test "" "SCRIPT;LABEL;TIMEOUT" "ARGS;PROPERTIES")
    if(NOT test_LABEL)
        message(FATAL_ERROR "anyps5_add_python_test(${name}) needs a LABEL")
    endif()
    anyps5_require_test_label(${test_LABEL})
    if(NOT Python3_Interpreter_FOUND)
        return()
    endif()
    if(NOT test_TIMEOUT)
        anyps5_default_test_timeout(${test_LABEL} test_TIMEOUT)
    endif()
    add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE} ${test_SCRIPT} ${test_ARGS})
    anyps5_set_test_properties(${name} ${test_LABEL} "${test_TIMEOUT}" ${test_PROPERTIES})
endfunction()
