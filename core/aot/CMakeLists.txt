cmake_minimum_required(VERSION 3.22.1)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AnyPS5LeafCompiler LANGUAGES C CXX)
    option(BUILD_TESTING "Build the leaf compiler tests" ON)
    if(BUILD_TESTING)
        enable_testing()
    endif()
else()
    enable_language(C)
endif()

get_filename_component(anyps5Root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(llvmSource "${anyps5Root}/3rdparty/llvm-project/llvm")
if(NOT EXISTS "${llvmSource}/CMakeLists.txt")
    message(FATAL_ERROR "Initialize 3rdparty/llvm-project to build the leaf compiler")
endif()
set(LLVM_TARGETS_TO_BUILD X86 CACHE STRING "" FORCE)
set(LLVM_ENABLE_PROJECTS "" CACHE STRING "" FORCE)
set(LLVM_ENABLE_RUNTIMES "" CACHE STRING "" FORCE)
foreach(feature IN ITEMS TESTS EXAMPLES BENCHMARKS DOCS TOOLS UTILS)
    set(LLVM_INCLUDE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(feature IN ITEMS ZLIB ZSTD LIBXML2 CURL LIBEDIT TERMINFO LIBPFM Z3_SOLVER FFI HTTPLIB)
    set(LLVM_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLVM_ENABLE_BINDINGS OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Backtrace TRUE)
set(LLVM_ENABLE_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${llvmSource}" "${CMAKE_CURRENT_BINARY_DIR}/llvm" EXCLUDE_FROM_ALL)
add_subdirectory(compiler)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
