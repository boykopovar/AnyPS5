include(${CMAKE_CURRENT_SOURCE_DIR}/Jemalloc.cmake)

add_library(libSceJemspace SHARED EXCLUDE_FROM_ALL Export.cpp GuestPages.c GuestPages.cpp)
set_target_properties(libSceJemspace PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
set_source_files_properties(GuestPages.c GuestPages.cpp PROPERTIES COMPILE_OPTIONS "-fvisibility=hidden")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set_source_files_properties(GuestPages.c PROPERTIES COMPILE_DEFINITIONS _GNU_SOURCE)
endif()
set_property(TARGET libSceJemspace PROPERTY NO_SYSTEM_FROM_IMPORTED TRUE)
target_include_directories(libSceJemspace PRIVATE ${LIBS_INCLUDE_DIR})
target_include_directories(libSceJemspace PRIVATE ${jemspaceSource}/include)
target_include_directories(libSceJemspace BEFORE PRIVATE ${jemspaceBuild}/include)
target_link_libraries(libSceJemspace PRIVATE jemspace_allocator)
set(jemspaceExports
    sceJemspaceCalloc sceJemspaceCreate sceJemspaceDestroy sceJemspaceFree sceJemspaceGetStats
    sceJemspaceMalloc sceJemspaceMallocUsableSize sceJemspaceMemalign sceJemspacePosixMemalign
    sceJemspaceRealloc sceJemspaceReallocalign sceJemspacePurge sceJemspacePrintStats)
if(APPLE)
    set(exportList "")
    foreach(symbol IN LISTS jemspaceExports)
        string(APPEND exportList "_${symbol}\n")
    endforeach()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/jemspace.exports" "${exportList}")
    target_link_options(libSceJemspace PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_BINARY_DIR}/jemspace.exports")
elseif(WIN32)
    set(exportList "EXPORTS\n")
    foreach(symbol IN LISTS jemspaceExports)
        string(APPEND exportList "${symbol}\n")
    endforeach()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/jemspace.def" "${exportList}")
    target_sources(libSceJemspace PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/jemspace.def")
    target_link_options(libSceJemspace PRIVATE LINKER:--exclude-libs,ALL)
else()
    target_link_options(libSceJemspace PRIVATE LINKER:--exclude-libs,ALL)
endif()
configure_windows_unwind(libSceJemspace)

if(BUILD_TESTING)
    add_test_executable(guest_jemspace_stats_tests ${LIBS_INCLUDE_DIR}/tests/GuestJemspaceStats.cpp)
    target_include_directories(guest_jemspace_stats_tests PRIVATE ${LIBS_INCLUDE_DIR})
    target_link_libraries(guest_jemspace_stats_tests PRIVATE libSceJemspace libc)
    configure_windows_unwind(guest_jemspace_stats_tests)
    add_test(NAME guest_jemspace_stats COMMAND guest_jemspace_stats_tests)
    add_test_executable(guest_jemspace_allocator_tests ${LIBS_INCLUDE_DIR}/tests/GuestJemspaceAllocator.cpp)
    target_include_directories(guest_jemspace_allocator_tests PRIVATE ${LIBS_INCLUDE_DIR})
    target_link_libraries(guest_jemspace_allocator_tests PRIVATE libSceJemspace libkernel libc Threads::Threads)
    configure_windows_unwind(guest_jemspace_allocator_tests)
    add_test(NAME guest_jemspace_allocator COMMAND guest_jemspace_allocator_tests)
    set_tests_properties(guest_jemspace_allocator PROPERTIES TIMEOUT 25)
    if(WIN32)
        set_tests_properties(guest_jemspace_stats guest_jemspace_allocator PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceJemspace>")
    endif()
endif()
