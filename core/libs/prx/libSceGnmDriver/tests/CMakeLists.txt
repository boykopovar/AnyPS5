add_test_executable(guest_gnm_draw_writers_tests ${LIBS_INCLUDE_DIR}/tests/GuestGnmDrawWriters.cpp)
target_include_directories(guest_gnm_draw_writers_tests PRIVATE ${LIBS_INCLUDE_DIR})
target_link_libraries(guest_gnm_draw_writers_tests PRIVATE libc libkernel ${CMAKE_DL_LIBS})
add_dependencies(guest_gnm_draw_writers_tests libSceGnmDriver libSceGnmDriverForNeoMode)
configure_windows_unwind(guest_gnm_draw_writers_tests)
foreach(drawProvider IN ITEMS normal neo)
    if(drawProvider STREQUAL "normal")
        set(drawLibrary libSceGnmDriver)
    else()
        set(drawLibrary libSceGnmDriverForNeoMode)
    endif()
    foreach(drawMode IN ITEMS base neo json none)
        set(drawTest guest_gnm_draw_${drawProvider}_${drawMode})
        add_test(NAME ${drawTest} COMMAND guest_gnm_draw_writers_tests ${drawProvider} ${drawMode} $<TARGET_FILE:${drawLibrary}>)
        set_tests_properties(${drawTest} PROPERTIES TIMEOUT 20)
        if(WIN32)
            set_tests_properties(${drawTest} PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:libc>")
        endif()
    endforeach()
endforeach()
