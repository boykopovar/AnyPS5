function(configure_quickjs_backend)
    set(BUILD_SHARED_LIBS OFF)
    set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory(${CMAKE_SOURCE_DIR}/3rdparty/quickjs ${CMAKE_BINARY_DIR}/3rdparty/quickjs EXCLUDE_FROM_ALL)
endfunction()
configure_quickjs_backend()
foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(libraries qjs ${property})
    if(libraries)
        list(REMOVE_ITEM libraries windows_seh_personalities)
        set_property(TARGET qjs PROPERTY ${property} "${libraries}")
    endif()
endforeach()
set_target_properties(qjs PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
if(MINGW OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(qjs INTERFACE LINKER:--exclude-libs,libqjs.a)
endif()
install(FILES ${CMAKE_SOURCE_DIR}/3rdparty/quickjs/LICENSE DESTINATION licenses/quickjs)

file(GLOB nkWebKitSources CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(libSceNKWebKit SHARED EXCLUDE_FROM_ALL Export.cpp DateParsing.cpp CharactersToNumber.cpp ${nkWebKitSources})
target_include_directories(libSceNKWebKit PRIVATE ${LIBS_INCLUDE_DIR})
target_link_libraries(libSceNKWebKit PUBLIC libSceJsc PRIVATE qjs libc)
configure_windows_unwind(libSceNKWebKit)
