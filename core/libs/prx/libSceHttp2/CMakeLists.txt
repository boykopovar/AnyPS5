function(configure_http2_backend)
    set(ENABLE_LIB_ONLY ON)
    set(ENABLE_DOC OFF)
    set(BUILD_SHARED_LIBS OFF)
    set(BUILD_STATIC_LIBS ON)
    foreach(package OpenSSL Libngtcp2 Libnghttp3 Systemd Jansson Libevent LibXml2 Jemalloc)
        set(CMAKE_DISABLE_FIND_PACKAGE_${package} TRUE)
    endforeach()
    add_subdirectory(${CMAKE_SOURCE_DIR}/3rdparty/nghttp2 ${CMAKE_BINARY_DIR}/3rdparty/nghttp2 EXCLUDE_FROM_ALL)
endfunction()
configure_http2_backend()
foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(libraries nghttp2_static ${property})
    if(libraries)
        list(REMOVE_ITEM libraries windows_seh_personalities)
        set_property(TARGET nghttp2_static PROPERTY ${property} "${libraries}")
    endif()
endforeach()
set_target_properties(nghttp2_static PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
if(MINGW OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(nghttp2_static INTERFACE LINKER:--exclude-libs,libnghttp2.a)
endif()
install(FILES ${CMAKE_SOURCE_DIR}/3rdparty/nghttp2/COPYING DESTINATION licenses/nghttp2)
