function(run_report_stamp_components root)
    foreach(component libkernel libSceVideoOut libSceAgcDriver)
        if(TARGET ${component})
            target_include_directories(${component} PRIVATE ${root})
            add_dependencies(${component} run_report_revision)
        endif()
    endforeach()
endfunction()
