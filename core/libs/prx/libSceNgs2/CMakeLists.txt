        set_tests_properties(ngs2_core_custom_safety_${safetyCase} PROPERTIES TIMEOUT 10)
        if(WIN32)
            set_tests_properties(ngs2_core_custom_safety_${safetyCase} PROPERTIES ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceNgs2>")
        endif()
    endforeach()
    foreach(ngs2Test IN ITEMS Gain Routing)
        string(TOLOWER ${ngs2Test} testName)
        add_test_executable(ngs2_core_${testName}_tests ${CMAKE_CURRENT_SOURCE_DIR}/../libSceNgs2.native/tests/Ngs2${ngs2Test}.cpp)
        target_include_directories(ngs2_core_${testName}_tests PRIVATE ${LIBS_INCLUDE_DIR} ${CMAKE_SOURCE_DIR}/3rdparty/LibAtrac9/C/src)
        target_link_libraries(ngs2_core_${testName}_tests PRIVATE libSceNgs2 atrac9 libkernel libc)
        configure_windows_unwind(ngs2_core_${testName}_tests)
        add_test(NAME ngs2_core_${testName} COMMAND ngs2_core_${testName}_tests)
        if(WIN32)
            set_tests_properties(ngs2_core_${testName} PROPERTIES ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceNgs2>")
        endif()
    endforeach()
    add_test_executable(ngs2_core_render_tests ${CMAKE_CURRENT_SOURCE_DIR}/../libSceNgs2.native/tests/Ngs2Render.cpp)
    target_include_directories(ngs2_core_render_tests PRIVATE ${LIBS_INCLUDE_DIR} ${CMAKE_SOURCE_DIR}/3rdparty/LibAtrac9/C/src)
    target_link_libraries(ngs2_core_render_tests PRIVATE libSceNgs2 atrac9 libkernel libc)
    configure_windows_unwind(ngs2_core_render_tests)
    add_test(NAME ngs2_core_render COMMAND ngs2_core_render_tests)
    if(WIN32)
        set_tests_properties(ngs2_core_render PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceNgs2>")
    endif()
    add_sce_ngs2_test(libSceNgs2 ngs2_core_pcm Ngs2Pcm.cpp)
    add_sce_ngs2_test(libSceNgs2 ngs2_core_ps4_voice_state Ngs2Ps4VoiceState.cpp)
endif()
