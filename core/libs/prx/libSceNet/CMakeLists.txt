add_library(libSceNet SHARED EXCLUDE_FROM_ALL Export.cpp)
target_include_directories(libSceNet PRIVATE ${LIBS_INCLUDE_DIR})
configure_windows_unwind(libSceNet)

if(BUILD_TESTING)
    add_test_executable(guest_sce_net_tests ${LIBS_INCLUDE_DIR}/tests/GuestSceNet.cpp)
    target_include_directories(guest_sce_net_tests PRIVATE ${LIBS_INCLUDE_DIR})
    target_link_libraries(guest_sce_net_tests PRIVATE libSceNet libSceNetCtl libkernel libc)
    configure_windows_unwind(guest_sce_net_tests)
    add_test(NAME guest_sce_net COMMAND guest_sce_net_tests)
    set_tests_properties(guest_sce_net PROPERTIES TIMEOUT 20)
    if(WIN32)
        add_test_executable(guest_receive_shutdown_tests ${LIBS_INCLUDE_DIR}/tests/GuestReceiveShutdown.cpp)
        target_include_directories(guest_receive_shutdown_tests PRIVATE ${LIBS_INCLUDE_DIR})
        target_link_libraries(guest_receive_shutdown_tests PRIVATE libSceNet libkernel libc)
        configure_windows_unwind(guest_receive_shutdown_tests)
        set(receiveKinds recv recvfrom recvmsg)
        foreach(api kernel sce)
            foreach(kind IN LISTS receiveKinds)
                list(FIND receiveKinds ${kind} kindIndex)
                add_test(NAME guest_${api}_${kind}_shutdown COMMAND guest_receive_shutdown_tests ${api} ${kindIndex})
                set_tests_properties(guest_${api}_${kind}_shutdown PROPERTIES TIMEOUT 20
                    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceNet>")
            endforeach()
        endforeach()
    endif()
    if(WIN32)
        set_tests_properties(guest_sce_net PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceNet>")
    endif()
endif()
